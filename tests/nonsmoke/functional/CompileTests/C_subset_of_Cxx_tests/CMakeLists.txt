function(compile_test_in_binary_dir input_file label)
  get_filename_component(test_name ${input_file} NAME_WE)

  configure_file(
    ${CMAKE_CURRENT_SOURCE_DIR}/../Cxx_tests/${input_file}
    ${CMAKE_CURRENT_BINARY_DIR}/${test_name}.c
    COPYONLY)

  add_test(
    NAME ${label}_${test_name}
    COMMAND ${translator}
      ${ROSE_FLAGS} ${ROSE_INCLUDE_FLAGS}
      -I${CMAKE_CURRENT_SOURCE_DIR}/../Cxx_tests
      -c ${CMAKE_CURRENT_BINARY_DIR}/${test_name}.c)
  set(_labels ${label})
  set(_disable_test FALSE)
  if(DEFINED ROSE_DISABLED_TESTCODES AND input_file IN_LIST ROSE_DISABLED_TESTCODES)
    list(APPEND _labels disabled)
    if(NOT REX_ENABLE_DISABLED_TESTS)
      set(_disable_test TRUE)
    endif()
  endif()
  if(DEFINED ROSE_FAILING_TESTCODES AND input_file IN_LIST ROSE_FAILING_TESTCODES)
    list(APPEND _labels known_fail)
    if(NOT REX_ENABLE_FAILING_TESTS)
      list(APPEND _labels disabled)
      set(_disable_test TRUE)
    endif()
  endif()
  list(REMOVE_DUPLICATES _labels)
  set_tests_properties(${label}_${test_name} PROPERTIES LABELS "${_labels}")
  if(_disable_test)
    set_tests_properties(${label}_${test_name} PROPERTIES DISABLED TRUE)
  endif()
endfunction()

foreach(file_to_test ${files_to_test})
  compile_test_in_binary_dir(${file_to_test} CSUBSETOFCXXTEST)
endforeach()
