function(fortran_test input_file)
  add_test(
    NAME FORTRANTEST_${input_file}_translation
    COMMAND testTranslator ${ROSE_FLAGS} ${FORTRAN_FLAGS}
            -c ${CMAKE_CURRENT_SOURCE_DIR}/${input_file})
  add_test(
    NAME FORTRANTEST_${input_file}_graph_generation
    COMMAND testGraphGeneration ${ROSE_FLAGS} ${FORTRAN_FLAGS}
            -c ${CMAKE_CURRENT_SOURCE_DIR}/${input_file})
  add_test(
    NAME FORTRANTEST_${input_file}_token_generation
    COMMAND testTokenGeneration ${ROSE_FLAGS} ${FORTRAN_FLAGS}
            -c ${CMAKE_CURRENT_SOURCE_DIR}/${input_file})
  set(_labels FORTRANTEST)
  set(_disable_test FALSE)
  if(DEFINED ROSE_DISABLED_TESTCODES AND input_file IN_LIST ROSE_DISABLED_TESTCODES)
    list(APPEND _labels disabled)
    if(NOT REX_ENABLE_DISABLED_TESTS)
      set(_disable_test TRUE)
    endif()
  endif()
  list(REMOVE_DUPLICATES _labels)
  set_tests_properties(FORTRANTEST_${input_file}_translation PROPERTIES LABELS "${_labels}")
  set_tests_properties(FORTRANTEST_${input_file}_graph_generation PROPERTIES DEPENDS FORTRANTEST_${input_file}_translation LABELS "${_labels}")
  set_tests_properties(FORTRANTEST_${input_file}_token_generation PROPERTIES DEPENDS FORTRANTEST_${input_file}_graph_generation LABELS "${_labels}")

  if(_disable_test)
    set_tests_properties(FORTRANTEST_${input_file}_translation PROPERTIES DISABLED TRUE)
    set_tests_properties(FORTRANTEST_${input_file}_graph_generation PROPERTIES DISABLED TRUE)
    set_tests_properties(FORTRANTEST_${input_file}_token_generation PROPERTIES DISABLED TRUE)
  endif()
endfunction()
